# Example only — renamed .txt so Fleet never applies it. Create the real one
# out-of-band from the files in ~/Developer/keys/jev/ (never commit values):
#
#   kubectl -n ai create secret generic jev-mcp \
#     --from-file=api_key=$HOME/Developer/keys/jev/claude.key \
#     --from-file=path_secret=$HOME/Developer/keys/jev/mcp-path-secret
#
# --from-file is safe here (unlike 10-hermes): the server reads both values via
# *_FILE and strips whitespace. path_secret is 48 hex chars from
# `openssl rand -hex 24`; the MCP endpoint becomes
# https://jev-mcp.ash4d.com/<path_secret>/mcp and that URL is the credential.
apiVersion: v1
kind: Secret
metadata:
  name: jev-mcp
  namespace: ai
type: Opaque
stringData:
  api_key: <TYPESAFE-API-KEY>
  path_secret: <48-HEX-RANDOM>
