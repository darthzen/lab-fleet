# Create the real secret out-of-band; never commit the actual token.
# A Full FOSSA API token (app.fossa.com -> Settings -> Integrations -> API):
#
#   kubectl -n ai create secret generic fossa-mcp-token \
#     --from-literal=token="$(tr -d '\r\n' < ~/Developer/keys/fossa/token.txt)"
#
# --from-file would keep the file's trailing newline in the value. The server
# strips surrounding whitespace from the token itself, so either form works, but
# the tr form keeps the stored secret byte-exact.
#
# Rotating: recreate the secret, then restart the deployment — the token is read
# once at process start.
#
#   kubectl -n ai delete secret fossa-mcp-token
#   kubectl -n ai create secret generic fossa-mcp-token --from-literal=token=<NEW>
#   kubectl -n ai rollout restart deploy/fossa-mcp
