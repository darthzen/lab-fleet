# The queue database's password. Firecrawl's NuQ queue lives in this Postgres;
# it holds in-flight job state, not crawl results. The Deployment, the API, the
# worker and the nuq-worker all read this one key.
#
# It is a throwaway — the database is ClusterIP-only inside the `firecrawl`
# namespace, has no persistence (see nuq-postgres.yaml), and losing it costs
# in-flight jobs and nothing else. It still lives in a Secret rather than
# inline, because this repo is public and the no-values-in-git rule does not
# carve out exceptions for uninteresting values.
#
# Seed it before enabling the Fleet path:
#
#   kubectl --context sdf1 -n firecrawl create secret generic firecrawl-nuq-postgres \
#     --from-literal=POSTGRES_PASSWORD="$(openssl rand -hex 24)"
#
# Rotating it means restarting all four Deployments AND wiping the Postgres
# pod (the password is baked into the database on first init, so a new
# POSTGRES_PASSWORD against an existing data dir authenticates with the OLD
# one). Since the data dir is an emptyDir, deleting the nuq-postgres pod is
# the whole rotation procedure.

apiVersion: v1
kind: Secret
metadata:
  name: firecrawl-nuq-postgres
  namespace: firecrawl
type: Opaque
stringData:
  POSTGRES_PASSWORD: "REPLACE_ME"
