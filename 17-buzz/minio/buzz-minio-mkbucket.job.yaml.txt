# ONE-TIME BOOTSTRAP — not part of the bundle.
#
# Renamed to .txt so Fleet never applies it, the same convention as
# 09-mcp/github-mcp-token.secret.example.yaml.txt.
#
# This Job creates the buzz-media bucket. Last run at the 2026-11-12 rebuild
# (it runs once and is reaped), so committing it as a live manifest would
# make Fleet CREATE it -- not a no-op. Jobs are also near-immutable, so a
# Job inside a synced bundle fights Fleet on every subsequent change.
#
# Run by hand if the bucket ever needs recreating:
#   kubectl -n buzz create -f buzz-minio-mkbucket.job.yaml.txt
apiVersion: batch/v1
kind: Job
metadata:
  name: buzz-minio-mkbucket
spec:
  backoffLimit: 6
  ttlSecondsAfterFinished: 300
  template:
    spec:
      restartPolicy: OnFailure
      containers:
      - name: mc
        image: pgsty/mc:RELEASE.2026-09-16T00-00-00Z@sha256:cfc83108c3abb371f8fb84d99c1fdc88f8c237e022409b0081fb7c0a3be634dd
        envFrom:
        - secretRef:
            name: buzz-minio-root
        command:
        - /bin/sh
        - -c
        args:
        - "set -e\nuntil mc alias set buzz http://buzz-minio:9000 \\\n      \"$MINIO_ROOT_USER\"\
          \ \"$MINIO_ROOT_PASSWORD\"; do\n  echo \"waiting for minio...\"; sleep 3\n\
          done\nmc mb --ignore-existing buzz/buzz-media\nmc ls buzz\n"
