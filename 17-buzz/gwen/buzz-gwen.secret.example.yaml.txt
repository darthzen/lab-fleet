# Example only — renamed .txt so Fleet never applies it. Create the real one
# out of band from the key files (never commit values; this repo is public):
#
#   BK=$(tr -d '\n' < ~/Developer/keys/buzz/gwen.key)
#   OR=$(tr -d '\n' < ~/Developer/keys/openrouter/gwen.key)
#   kubectl -n buzz create secret generic buzz-gwen \
#     --from-literal=BUZZ_PRIVATE_KEY="$BK" \
#     --from-literal=OPENROUTER_API_KEY="$OR"
#
# --from-literal with the newline stripped, NOT --from-file: both values land
# in env vars, and a trailing newline in OPENROUTER_API_KEY corrupts the
# Authorization header (cf. the hermes-webui incident, 10-hermes).
#
# BUZZ_PRIVATE_KEY is Gwen's Nostr secret key, 64 hex chars, from
#   kubectl exec -n buzz deploy/buzz -- buzz-admin generate-key
# The matching public key is ~/Developer/keys/buzz/gwen.pub and is what
# buzz-admin add-member takes. OPENROUTER_API_KEY is a key of Gwen's own, so
# her usage and limits stay apart from Hermes.
apiVersion: v1
kind: Secret
metadata:
  name: buzz-gwen
  namespace: buzz
type: Opaque
stringData:
  BUZZ_PRIVATE_KEY: <64-HEX-NOSTR-SECRET-KEY>
  OPENROUTER_API_KEY: <sk-or-v1-...>
