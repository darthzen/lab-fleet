# Example only — renamed .txt so Fleet never applies it. Create the real one
# out of band from the key files (never commit values; this repo is public):
#
#   kubectl -n buzz create secret generic gwen \
#     --from-literal=BUZZ_PRIVATE_KEY="$(tr -d '\n' < ~/Developer/keys/buzz/gwen.key)" \
#     --from-literal=OPENROUTER_API_KEY="$(tr -d '\n' < ~/Developer/keys/openrouter/gwen.key)" \
#     --from-literal=HINDSIGHT_API_KEY="$(tr -d '\n' < ~/Developer/keys/vectorize/hindsight-api-key)"
#
# --from-literal with the newline stripped, NOT --from-file: the values land in
# env vars, and a trailing newline corrupts an Authorization header (cf. the
# hermes-webui incident, 10-hermes).
#
# BUZZ_PRIVATE_KEY is Gwen's Nostr secret key, 64 hex chars. The matching
# public key is ~/Developer/keys/buzz/gwen.pub and is what buzz-admin
# add-member takes. OPENROUTER_API_KEY is a key of Gwen's own, so her usage and
# limits stay apart from Hermes; it also pays for Jev routing (OpenRouter
# Decisions API). HINDSIGHT_API_KEY is Rick's Hindsight key: Gwen reads and
# writes bank `default` (Rick's memory) and bank `gwen` (her own).
apiVersion: v1
kind: Secret
metadata:
  name: gwen
  namespace: buzz
type: Opaque
stringData:
  BUZZ_PRIVATE_KEY: <64-HEX-NOSTR-SECRET-KEY>
  OPENROUTER_API_KEY: <sk-or-v1-...>
  HINDSIGHT_API_KEY: <hindsight-api-key>
