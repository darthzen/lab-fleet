# Example only — renamed .txt so Fleet never applies it. Create the real one
# out of band (never commit values; this repo is public):
#
#   kubectl -n buzz create secret generic workspace-mcp \
#     --from-literal=GOOGLE_OAUTH_CLIENT_ID="<client-id-from-Google-Cloud>" \
#     --from-literal=GOOGLE_OAUTH_CLIENT_SECRET="<client-secret-from-Google-Cloud>"
#
# GOOGLE_OAUTH_CLIENT_ID and GOOGLE_OAUTH_CLIENT_SECRET come from the Web
# application OAuth client in your Google Cloud project. The client must have
# the six APIs enabled: gmail, calendar-json, drive, docs, sheets, slides.
# The redirect URI in the client must be http://localhost:8000/oauth2callback
# (matching GOOGLE_OAUTH_REDIRECT_URI in the Deployment).
apiVersion: v1
kind: Secret
metadata:
  name: workspace-mcp
  namespace: buzz
type: Opaque
stringData:
  GOOGLE_OAUTH_CLIENT_ID: <google-oauth-client-id>
  GOOGLE_OAUTH_CLIENT_SECRET: <google-oauth-client-secret>
